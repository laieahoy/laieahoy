## A. Cezar
对于不同字符串 $s_1,s_2,...s_n$，因为是字典序顺序，所以我们要从高到低比较。现在，假设我们取出了两个排序后相邻的字符串 $s_p,s_q$（$s_p$ 在 $s_q$ 前面），比较到了 $x$ 这一位置上，则 $s_p,s_q$ 前面的 $x-1$ 位都相同，因为字典序排序是从高到低，若前面有不同的，那么后面的比较就失去了意义。那么钦定它们的第 $x$ 位不同，比如 $s_p[x]$ 为任意字母 $a$，$s_q[x]$ 为任意不为 $a$ 的字母 $b$，那么根据题目要求，字符 $a$ 在密钥中的排序一定先于 $b$，然后我们就可以跳出比较，进行下一次比较了。

比较完后，会发现我们得到了若干组的字母先后顺序，我们要做的就是给一个密钥序列，使得这若干组先后顺序能够全部被满足。

发现字母的先后顺序其实是一种**指向关系**，这种关系常常会被转化到图论中，通过**建立有向图**来解决问题，这道就是这样，我们将形如 $a$ 先于 $b$（$a$，$b$ 是任意不相同的两个字母） 这种字母关系转化 $a\to b$ 的一条有向边，那么就可以发现，判断是否能够构造出满足条件的密钥，实际就是在判建出的图中是否有环（即是否满足拓扑序），然后构造答案的话，如果有拓扑关系的就按先后顺序来，没有的就随意排列。而拓扑关系的排序恰好是 bfs 遍历时访问节点的顺序，于是可以判环时一并处理掉。

本题还需要考虑一种特殊情况，就是当一个字符串是上一个字符串的前缀时，比如 $s_p$ 为 $sjkdnfdfa$，而 $s_q$ 为$sjkdnf$，那么这种情况显然也是不合法的。

```cpp
#include<iostream>
#include<algorithm>
#include<string.h>
#include<queue>
using namespace std;

const int N=105;
int n;
string s[N];
int a[N];
int wx,ans[N],num[N];
int tot,cnt,in[N],res[N];
bool vis[N],flag[N];
int e[N<<1],ne[N<<1],h[N],idx;

inline void add(int u,int v)
{
    e[idx]=v;
    ne[idx]=h[u];
    h[u]=idx++;
}

int topp()
{
    queue<int> q;
    for(int i=0;i<26;i++){
        if(!in[i]&&vis[i]) q.push(i);
    }
    while(q.size()){
        int u=q.front();
        q.pop();
        flag[u]=1;
        ans[++wx]=u;
        num[wx]=u;
        for(int i=h[u];~i;i=ne[i]){
            int v=e[i];
            if(!(--in[v])){q.push(v);}
        }
    }
    if(wx!=cnt) return -1;
    return wx;
}

int main()
{
    memset(h,-1,sizeof h);
    cin>>n;
    for(int i=1;i<=n;i++) cin>>s[i];
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<n;i++){
        int i1=0,i2=0;
        bool f=0;
        while(i1<s[a[i]].size()&&i2<s[a[i+1]].size()){
            if(s[a[i]][i1]!=s[a[i+1]][i2]){
                f=1;
                add(s[a[i]][i1]-'a',s[a[i+1]][i2]-'a');
                if(!vis[s[a[i]][i1]-'a']) cnt++,vis[s[a[i]][i1]-'a']=1;
                if(!vis[s[a[i+1]][i2]-'a']) cnt++,vis[s[a[i+1]][i2]-'a']=1;
                in[s[a[i+1]][i2]-'a']++;
                break;
            }
            i1++,i2++;
        }
        if(!f&&s[a[i]].size()>s[a[i+1]].size()){
            puts("NE");
            return 0;
        }
    }
    int t=topp();
    if(t==-1){
        puts("NE");
        return 0;
    }
    sort(ans+1,ans+1+wx);
    for(int i=1;i<=wx;i++) res[num[i]]=ans[i];
    for(int i=0;i<26;i++){
        if(vis[i]&&!flag[i]){
            puts("NE");
            return 0;
        }
        if(!flag[i]) res[i]=i;
    }
    puts("DA");
    for(int i=0;i<26;i++) cout<<char(res[i]+'a');

    return 0;
}
```

## C. Poklon
完全莫队模板题，但是注意一下莫队的排序方式，不然会超时。

```cpp
#include<iostream>
#include<algorithm>
#include<string.h>
#include<cmath>
#include<vector>
using namespace std;

const int N=5e5+5;
int n,q,block,l,r;
int a[N],ans[N],cnt[N],num;
struct Node{
    int l,r,id;
}b[N];

inline int read()
{
    int f=1,res=0;
    char ch=getchar();
    while(ch<'0'||ch>'9'){
        if(ch=='-') f=-1;
        ch=getchar();
    }
    while(ch>='0'&&ch<='9'){
        res=res*10+ch-'0';
        ch=getchar();
    }
    return res*f;
}

inline void write(int x) {
    if (x>=10) write(x/10);
    putchar(x%10+'0');
}

bool cmp(Node a,Node b)
{
    int tx=a.l/block,ty=b.l/block;
    if(tx!=ty) return tx<ty;
    if(tx&1) return a.r<b.r;
    return a.r>b.r;
}

void add(int x)
{
    if(cnt[a[x]]==2) num--;
    if(cnt[a[x]]==1) num++;
    cnt[a[x]]++;
}

void del(int x)
{
    if(cnt[a[x]]==2) num--;
    if(cnt[a[x]]==3) num++;
    cnt[a[x]]--;
}

int main()
{
    n=read(),q=read();
    for(int i=1;i<=n;i++) a[i]=read();
    for(int i=1;i<=q;i++){
        b[i].l=read(),b[i].r=read();
        b[i].id=i;
    }
    block=n/(sqrt(q*2/3));
    vector<int> v;
    for(int i=1;i<=n;i++) v.push_back(a[i]);
    sort(v.begin(),v.end());
    v.erase(unique(v.begin(),v.end()),v.end());
    for(int i=1;i<=n;i++) a[i]=lower_bound(v.begin(),v.end(),a[i])-v.begin()+1;
    sort(b+1,b+1+q,cmp);
    l=r=0;
    for(int i=1;i<=q;i++){
        while(l<b[i].l) del(l++);
        while(r>b[i].r) del(r--);
        while(l>b[i].l) add(--l);
        while(r<b[i].r) add(++r);
        ans[b[i].id]=num;
    }
    for(int i=1;i<=q;i++){
        write(ans[i]);
        puts("");
    }

    return 0;
}
```
